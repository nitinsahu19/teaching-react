--------------------------------------------------- what is lazy loading ? -------------------------------------------------
=> Lazy loading in react renders the component or webpage only when the user needs it . This can reduce the web page loading time.
{syntax} -> const Todo = React.lazy(()=> import ("./components/Todo/Todo"));


---------------------------------------------- what is suspense ? -------------------------------------------------------
=> React suspense is a built-in React component thet wraps around another component to change its behaviour. fallback show the text when 
the content is loading. 
{syntax} -> 
import React, { Suspense } from 'react';

const About = React.lazy(() => import('./About'));      //this is lazy loading

function App() {
    return (
        <Suspense fallback={<div>Loading...</div>}>
            <About />
        </Suspense>
    );
}

export default App;

------------------------------------------ what is react memo? ---------------------------------------------------------
=> React memo is High Order component (HOC) that memoizes the passed in component along with value of its props. 
import React, { memo } from 'react';

const MyComponent = ({ name }) => {
    console.log('Rendering MyComponent');
    return <h1>Hello, {name}</h1>;
};

export default memo(MyComponent);

------------------------------------------------- what is usesearchParams? -----------------------------------------------
=> React Router ka hook hai jo hume URL ke query parameters (?key=value) read aur update karne deta hai.
import { useSearchParams } from "react-router-dom";

function MyComponent() {
  const [searchParams] = useSearchParams();
  const name = searchParams.get("name");

  return <h2>Hello, {name}</h2>;
}
export default MyComponent

----------------------------------------------what is usestate?----------------------------------------------------
=> Component ke andar data store karna.
import React, { useState } from "react";

function Counter() {
  const [count, setCount] = useState(0);

  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={() => setCount(count + 1)}>Increase</button>
    </div>
  );
}
export default Counter
